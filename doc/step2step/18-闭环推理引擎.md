### 1. 闭环推理引擎设计

```python
# backend/services/inference_engine.py
import asyncio
import time
from enum import Enum

class InferenceState(Enum):
    IDLE = "idle"
    RUNNING = "running"
    PAUSED = "paused"
    DONE = "done"

class InferenceEngine:
    """闭环推理引擎：驱动仿真+模型推理的闭环控制循环"""

    def __init__(self, sim_service, adapter_manager, config=None):
        self.sim_service = sim_service
        self.adapter_manager = adapter_manager
        self.state = InferenceState.IDLE
        self.speed = 1.0             # 播放速度倍率
        self.max_steps = 200          # 最大推理步数（200 步 = 4 秒 @ 50Hz）
        self.action_queue = asyncio.Queue(maxsize=10)
        self._inference_task = None
        # 结果
        self.final_error = None
        self.total_steps = 0
        self.collision_count = 0

    async def start(self):
        """启动闭环推理"""
        if self.state == InferenceState.RUNNING:
            return
        self.state = InferenceState.RUNNING
        self._inference_task = asyncio.create_task(self._run_loop())

    async def stop(self):
        """停止推理"""
        self.state = InferenceState.IDLE
        if self._inference_task:
            self._inference_task.cancel()

    async def _run_loop(self):
        """主闭环循环"""
        # 重置环境到任务初始状态
        await self._reset_environment()

        step = 0
        while step < self.max_steps and self.state == InferenceState.RUNNING:
            loop_start = time.time()

            # 1. 获取观测
            observation = await self._get_observation()

            # 2. 模型推理（线程池异步执行）
            action = await self._infer(observation)

            # 3. 执行动作
            await self._execute_action(action)

            # 4. 检测终止条件
            done, success, info = await self._check_termination()
            if done:
                await self._on_done(success, info)
                return

            # 5. 步进控制（50Hz / speed）
            step += 1
            self.total_steps = step
            elapsed = time.time() - loop_start
            target_dt = 0.02 / self.speed  # 20ms per step at 1x speed
            if elapsed < target_dt:
                await asyncio.sleep(target_dt - elapsed)

        # 超时
        await self._on_done(False, {"reason": "timeout", "steps": step})

    async def _get_observation(self) -> dict:
        """获取当前观测：图像 + 状态 + 指令"""
        loop = asyncio.get_event_loop()
        state = await loop.run_in_executor(
            self.sim_service.executor,
            lambda: self.sim_service.controller.get_state(),
        )
        images = await loop.run_in_executor(
            self.sim_service.executor,
            lambda: self.sim_service.controller.get_camera_images(),
        )
        return {
            "top_image": images["top"],
            "angle_image": images["angle"],
            "ee_pose": state["ee_pose"],
            "obj_pose": state["obj_pose"],
            "instruction": self.sim_service.current_instruction,
            "target_pose": state["target_pose"],
        }

    async def _infer(self, observation: dict) -> np.ndarray:
        """调用模型适配器进行单步推理"""
        loop = asyncio.get_event_loop()
        # 使用顶视图作为模型输入（也可拼接双视图）
        image = observation["top_image"]
        instruction = observation["instruction"]
        state = observation["ee_pose"]

        action = await loop.run_in_executor(
            self.sim_service.executor,
            lambda: self.adapter_manager.predict(image, instruction, state),
        )
        # 广播动作到前端
        await self.sim_service.manager.broadcast({
            "type": "inference_action",
            "action": action.tolist(),
            "model_name": self.adapter_manager.current_model_name,
            "timestamp": time.time(),
        })
        return action

    async def _execute_action(self, action: np.ndarray):
        """在仿真中执行动作"""
        # 将动作放入队列，由仿真步进线程平滑执行
        await self.action_queue.put(action)
        loop = asyncio.get_event_loop()
        # 执行一步仿真（内部会从队列取动作）
        await loop.run_in_executor(
            self.sim_service.executor,
            lambda: self.sim_service.controller.step(action),
        )

    async def _check_termination(self):
        """检查终止条件"""
        loop = asyncio.get_event_loop()
        done, success, info = await loop.run_in_executor(
            self.sim_service.executor,
            self.sim_service.controller.is_done,
        )
        return done, success, info

    async def _on_done(self, success, info):
        """推理结束后的清理和通知"""
        self.state = InferenceState.DONE
        self.final_error = info.get("final_error", None)
        await self.sim_service.manager.broadcast({
            "type": "done",
            "success": success,
            "final_error": self.final_error,
            "total_steps": self.total_steps,
            "collision_count": self.collision_count,
        })
```

### 2. 终止条件详解

| 条件 | 判断方式 | 结果 |
|------|----------|------|
| 成功 | 物体中心距目标 < 0.01m | `(True, True, {"final_error": error})` |
| 碰撞失败 | 物体与障碍物碰撞 | `(True, False, {"reason": "collision"})` |
| 限位失败 | 末端超出限位 | `(True, False, {"reason": "limit"})` |
| 超时 | 步数 > 200 | `(True, False, {"reason": "timeout"})` |
| 脱开 | 物体与推板脱离超过 0.05m 且持续 10 步 | `(True, False, {"reason": "lost_contact"})` |

### 3. 动作队列平滑执行

为确保 50Hz 仿真步进不受推理延迟影响：

```python
class ActionSmoother:
    """动作平滑器：对连续动作做插值，避免突变"""

    def __init__(self, interpolation_steps=4):
        self.prev_action = np.zeros(3)
        self.interpolation_steps = interpolation_steps

    def smooth(self, target_action: np.ndarray) -> list[np.ndarray]:
        """将目标动作拆分为多个平滑子步"""
        result = []
        for i in range(1, self.interpolation_steps + 1):
            t = i / self.interpolation_steps
            # 线性插值
            interp_action = self.prev_action + (target_action - self.prev_action) * t
            result.append(interp_action)
        self.prev_action = target_action
        return result
```

### 4. 速度控制

通过 `self.speed` 控制播放速率：
- **0.5x**：`target_dt = 0.04s`（25Hz），适合慢速观察
- **1x**：`target_dt = 0.02s`（50Hz），实时播放
- **2x**：`target_dt = 0.01s`（100Hz 逻辑步进，实际受限于 GPU），快速播放

## 注意事项
- 推理引擎以异步方式运行，不阻塞 FastAPI 事件循环
- 推理和仿真步进在同一线程执行（MuJoCo 线程安全），但推理本身可在线程池中执行
- 动作平滑可显著改善视觉效果，特别是在推理延迟较高时
- 首轮推理前应确保仿真已稳定（运行若干 idle 步使接触稳定）
- 记录每次推理的日志（步数、成功率、耗时等），便于后续分析模型性能

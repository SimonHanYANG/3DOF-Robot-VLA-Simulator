### 2. 脚本策略适配器

```python
@register_adapter("scripted")
class ScriptedAdapter(BaseModelAdapter):
    """脚本策略 — 回放预生成轨迹，模拟训练好的模型"""

    def __init__(self, config):
        super().__init__(config)
        self._trajectory = None
        self._step_idx = 0

    def load(self, checkpoint_path=None):
        pass

    def set_trajectory(self, actions: list[np.ndarray]):
        self._trajectory = actions
        self._step_idx = 0

    def predict(self, image, instruction, state=None):
        if self._trajectory is None or self._step_idx >= len(self._trajectory):
            return np.zeros(7)  # 无轨迹时返回零动作
        action = self._trajectory[self._step_idx]
        self._step_idx += 1
        return self._clip_action(action)
```

```python
def test_random_adapter():
    """测试 1: RandomAdapter 闭环"""
    sim = SimController(SCENE_PATH)

    print(f"Random: steps={step}, done={done}, info={info}")

def test_scripted_adapter():
    """测试 2: 脚本策略闭环"""
    sim = SimController(SCENE_PATH)
    grasp_planner = GraspPlanner(sim.model, sim.data)
    traj_gen = PickPlaceTrajectoryGenerator(sim, grasp_planner)

    sim.reset("cube", (0.1, 0.0, 0.03), (0.2, 0.1, 0.03))
    actions = traj_gen.generate("cube", (0.1, 0.0, 0.03), (0.2, 0.1, 0.03))

    adapter = ScriptedAdapter({})
    adapter.load("")
    adapter.set_trajectory(actions)

    for step in range(500):
        obs = sim.get_obs()
        action = adapter.predict(obs["top_image"], "pick the cube", obs["joint_positions"])
        sim.step(action)
        done, success, info = sim.is_done()
        if done:
            break

    print(f"Scripted: steps={step}, success={success}, info={info}")
    assert success, "Scripted policy should succeed"
```

### 4. 通过 WebSocket 的端到端测试

```python
async def test_ws_inference():
    """测试 3: WebSocket 端到端"""
    # 启动后端
    # WebSocket 连接
    ws = await connect("ws://localhost:8765/ws")

    # 设置任务
    await ws.send({"type": "set_object", "object_type": "cube", "position": [0.1, 0, 0.03]})
    await ws.send({"type": "set_target", "position": [0.2, 0.1]})

    # 加载随机模型
    await ws.send({"type": "load_model", "model_name": "random", "checkpoint_path": ""})

    # 开始推理
    await ws.send({"type": "start_inference", "speed": 2.0})

    # 等待 state_update 和 done 消息
    done_received = False
    while not done_received:
        msg = await ws.recv()
        if msg["type"] == "done":
            done_received = True

    print("WebSocket inference completed")
```

# S17 — 统一推理接口与模型适配器

**状态**：🔴 未开始
**阶段**：六、模型推理与可视化
**依赖**：S14/S15/S16 中至少一个模型训练完成（有可用 checkpoint）

## 目标
定义统一的模型推理接口（抽象适配器基类），为每个 VLA 模型实现适配器，使上层推理引擎（S18）无需关心具体模型差异。

## 前置条件
- 至少一个模型训练完成且有可用 checkpoint（如 S15 ACT 训练完成）

## 实现细节

### 1. 抽象适配器基类

```python
# models/adapters/base_adapter.py
from abc import ABC, abstractmethod
import numpy as np

class BaseModelAdapter(ABC):
    """VLA 模型推理适配器基类

    所有模型适配器需实现此接口，确保上层推理引擎统一调用。
    """

    def __init__(self, config: dict):
        self.config = config
        self.model = None
        self.device = config.get("device", "cuda")
        self.action_range = config.get("action_range", (-0.02, 0.02))

    @abstractmethod
    def load(self, checkpoint_path: str):
        """加载模型权重和配置"""

    @abstractmethod
    def predict(self, image: np.ndarray, instruction: str,
                state: np.ndarray | None = None) -> np.ndarray:
        """
        单步推理
        参数:
            image: (H, W, 3) RGB 图像 numpy 数组
            instruction: 任务指令字符串
            state: 当前末端位置 (x, y, z)，可选
        返回:
            action: (3,) 增量位移 (dx, dy, dz)，范围 [-0.02, 0.02]
        """

    @abstractmethod
    def unload(self):
        """释放模型资源（GPU 显存）"""

    def _clip_action(self, action: np.ndarray) -> np.ndarray:
        """将动作裁剪到安全范围"""
        low, high = self.action_range
        return np.clip(action, low, high)
```

### 2. 适配器注册机制

```python
# models/adapters/registry.py
_ADAPTER_REGISTRY: dict[str, type[BaseModelAdapter]] = {}

def register_adapter(name: str):
    """装饰器：注册模型适配器"""
    def decorator(cls):
        _ADAPTER_REGISTRY[name] = cls
        return cls
    return decorator

def get_adapter(name: str) -> type[BaseModelAdapter]:
    """根据模型名称获取适配器类"""
    if name not in _ADAPTER_REGISTRY:
        raise ValueError(f"未知模型类型: {name}，已知: {list(_ADAPTER_REGISTRY.keys())}")
    return _ADAPTER_REGISTRY[name]

def create_adapter(model_name: str, config: dict) -> BaseModelAdapter:
    """工厂函数：创建适配器实例并加载模型"""
    adapter_cls = get_adapter(model_name)
    adapter = adapter_cls(config)
    adapter.load(config["checkpoint_path"])
    return adapter
```

### 3. OpenVLA 适配器

```python
# models/adapters/openvla_adapter.py
@register_adapter("openvla")
class OpenVLAAdapter(BaseModelAdapter):
    """OpenVLA 模型推理适配器"""

    def load(self, checkpoint_path: str):
        """加载微调后的 OpenVLA checkpoint"""
        # 加载模型和 processor
        self.model = OpenVLAModel.from_pretrained(checkpoint_path)
        self.processor = OpenVLAProcessor.from_pretrained(checkpoint_path)
        self.model.to(self.device)
        self.model.eval()
        # 动作解码参数
        self.num_bins = self.config.get("num_bins", 256)

    def predict(self, image, instruction, state=None):
        """图像+指令 → 增量动作 (dx, dy, dz)"""
        # 预处理
        inputs = self.processor(
            images=image,
            text=instruction,
            return_tensors="pt",
        ).to(self.device)

        # 推理
        with torch.no_grad():
            outputs = self.model.generate(**inputs)

        # 解码 action tokens → 连续动作
        action_tokens = outputs.action_tokens  # 具体字段名取决于模型
        action = self._decode_tokens(action_tokens)
        return self._clip_action(action)

    def _decode_tokens(self, tokens) -> np.ndarray:
        """将离散 token 解码为连续值"""
        # 与 S14 中的编码方式对应
        actions = []
        for dim_tokens in tokens.reshape(3, -1):
            bin_id = dim_tokens.mode().values.item()  # 多数投票
            value = (bin_id / self.num_bins) * 0.04 - 0.02
            actions.append(value)
        return np.array(actions)

    def unload(self):
        del self.model
        torch.cuda.empty_cache()
```

### 4. ACT 适配器

```python
# models/adapters/act_adapter.py
@register_adapter("act")
class ACTAdapter(BaseModelAdapter):
    """ACT 模型推理适配器"""

    def load(self, checkpoint_path: str):
        """加载训练好的 ACT checkpoint"""
        checkpoint = torch.load(checkpoint_path, map_location=self.device)
        self.model = ACTModel(
            action_dim=3,
            chunk_len=self.config.get("chunk_len", 100),
            hidden_dim=self.config.get("hidden_dim", 256),
        )
        self.model.load_state_dict(checkpoint["model_state_dict"])
        self.model.to(self.device)
        self.model.eval()
        # 动作块缓存
        self._action_buffer = []  # 尚未执行的预测动作
        self._step_counter = 0

    def predict(self, image, instruction, state=None):
        """图像+末端位置 → 增量动作序列 → 返回当前步动作"""
        # ACT 每 N 步推理一次（如 10 步），缓存动作块
        if len(self._action_buffer) == 0 or self._step_counter >= 10:
            # 需要重新推理
            image_tensor = self._preprocess_image(image).to(self.device)
            state_tensor = torch.tensor(state, dtype=torch.float32).to(self.device)

            with torch.no_grad():
                action_chunk = self.model(
                    image_tensor.unsqueeze(0),
                    state_tensor.unsqueeze(0),
                )  # (1, chunk_len, 3)

            self._action_buffer = action_chunk[0].cpu().numpy().tolist()
            self._step_counter = 0

        # 取缓存中的下一个动作
        action = np.array(self._action_buffer.pop(0))
        self._step_counter += 1

        # 如果使用 temporal ensemble，此处对多个预测取加权平均
        return self._clip_action(action)

    def _preprocess_image(self, image: np.ndarray) -> torch.Tensor:
        """图像预处理：resize, normalize"""
        import torchvision.transforms as T
        transform = T.Compose([
            T.ToTensor(),
            T.Resize((224, 224)),
            T.Normalize(mean=[0.485, 0.456, 0.406],
                       std=[0.229, 0.224, 0.225]),
        ])
        return transform(image)
```

### 5. π0/π0.5 适配器

```python
# models/adapters/pi0_adapter.py
@register_adapter("pi0")
class Pi0Adapter(BaseModelAdapter):
    """π0/π0.5 模型推理适配器"""

    def load(self, checkpoint_path: str):
        """加载微调后的 π0 checkpoint"""
        self.model = PiZeroModel.from_pretrained(checkpoint_path)
        self.model.to(self.device)
        self.model.eval()

    def predict(self, image, instruction, state=None):
        """多模态输入 → 连续动作"""
        inputs = {
            "image": image,
            "instruction": instruction,
            "state": state,
        }
        with torch.no_grad():
            # π0 使用扩散模型，推理可能需多步去噪
            action = self.model.predict_action(**inputs)
        return self._clip_action(action)

    def unload(self):
        del self.model
        torch.cuda.empty_cache()
```

### 6. 适配器管理器

```python
# models/adapters/manager.py
class AdapterManager:
    """管理模型的加载、切换、推理"""

    def __init__(self):
        self.current_adapter: BaseModelAdapter | None = None
        self.current_model_name: str | None = None

    def load_model(self, model_name: str, checkpoint_path: str, **config):
        """切换/加载模型"""
        # 先卸载当前模型
        if self.current_adapter is not None:
            self.current_adapter.unload()
        # 加载新模型
        config["checkpoint_path"] = checkpoint_path
        self.current_adapter = create_adapter(model_name, config)
        self.current_model_name = model_name

    def predict(self, image, instruction, state=None):
        """代理到当前加载的模型"""
        if self.current_adapter is None:
            raise RuntimeError("尚未加载模型")
        return self.current_adapter.predict(image, instruction, state)

    def unload(self):
        if self.current_adapter:
            self.current_adapter.unload()
            self.current_adapter = None
```

## 验证标准
- [ ] 每种适配器可加载对应的训练 checkpoint 并成功推理
- [ ] 推理输出 shape 统一为 (3,) 的增量位移，范围在 [-0.02, 0.02]
- [ ] 适配器切换：加载 OpenVLA → 卸载 → 加载 ACT → 正常运行
- [ ] 单步推理耗时满足实时性要求（OpenVLA < 100ms, ACT < 20ms, π0.5 < 100ms）
- [ ] `AdapterManager` 可通过 `model_name` 字符串动态创建适配器
- [ ] 推理时 GPU 显存使用稳定（无泄漏）
- [ ] 相同输入下多次推理输出一致（seed 固定时）

## 注意事项
- 不同模型对图像预处理的要求可能不同（归一化参数、尺寸等），在各自适配器中独立处理
- ACT 的动作块缓存策略影响推理平滑性——每 10 步推理一次，在步间顺序消耗缓存
- 适配器的 `predict` 方法是同步的，在 S18 中会在线程池中异步调用，避免阻塞事件循环
- 模型加载（`load`）可能耗时数秒，前端应显示加载进度
- 适配器需处理极端情况：纯黑图像、空指令等，返回零动作或报明确错误

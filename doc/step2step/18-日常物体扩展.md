### 3. 抓取策略更新

```python
def _get_grasp_strategy(self, obj_type: str):
    """获取各物体类型的抓取策略"""
    strategies = {
        # 几何体：顶部垂直抓取
        "cube": {"approach": "top_down", "offset": 0.0},
        "sphere": {"approach": "top_down", "offset": 0.0},
        "cylinder": {"approach": "top_down", "offset": 0.0},
        # 日常物体：形状特定策略
        "cup": {
            "approach": "top_down",
            "grasp_point": "rim",  # 抓杯沿
            "finger_width": 0.06,  # 杯口直径
            "offset": 0.02,        # 稍高于杯沿
        },
        "bottle": {
            "approach": "top_down",
            "grasp_point": "body",  # 抓瓶身中部
            "finger_width": 0.03,   # 瓶身直径
            "offset": 0.0,
        },
        "bowl": {
            "approach": "top_down",
            "grasp_point": "rim",   # 抓碗沿
            "finger_width": 0.08,   # 碗口直径
            "offset": 0.02,
        },
    }
    return strategies.get(obj_type, strategies["cube"])
```

def test_daily_objects():

        sim.reset(obj_type, (0.1, 0.0), (0.2, 0.1))
        grasp = planner.plan_grasp(obj_type, sim.get_obj_pos(obj_type), ...)
        assert grasp is not None, f"Failed to plan grasp for {obj_type}"

## 注意事项
- 简单参数化 mesh 可能碰撞几何不够精确，影响抓取稳定性
- 杯子和碗是薄壁结构，mesh 碰撞需使用 convex hull 或简化
- 瓶子的重心偏高，抓取后可能倾倒——需验证质量分布参数
- 日常物体的抓取成功率通常低于几何体，数据筛选标准可适当放宽

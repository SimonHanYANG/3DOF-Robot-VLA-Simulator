## 实现细节

### 1. 抽象适配器基类（7D）

```python
class BaseModelAdapter(ABC):
    """VLA 模型推理适配器基类 — 7D 动作输出"""

    def __init__(self, config: dict):
        self.config = config
        self.device = config.get("device", "cuda")
        # 7D 动作范围
        self.action_low = np.array([
            -0.02, -0.02, -0.02,      # dx, dy, dz
            -0.1, -0.1, -0.1,         # droll, dpitch, dyaw
            0.0                        # gripper (closed)
        ])
        self.action_high = np.array([
            0.02, 0.02, 0.02,
            0.1, 0.1, 0.1,
            1.0                        # gripper (open)
        ])

    @abstractmethod
    def load(self, checkpoint_path: str):
        """加载模型权重"""

    @abstractmethod
    def predict(self, image: np.ndarray, instruction: str,
                state: np.ndarray | None = None) -> np.ndarray:
        """
        单步推理
        返回: action (7,) = (dx,dy,dz,droll,dpitch,dyaw,gripper)
        """

    @abstractmethod
    def unload(self):
        """释放资源"""

    def _clip_action(self, action: np.ndarray) -> np.ndarray:
        return np.clip(action, self.action_low, self.action_high)
```

### 2. 适配器注册机制

```python
_ADAPTER_REGISTRY: dict[str, type[BaseModelAdapter]] = {}

def register_adapter(name: str):
    def decorator(cls):
        _ADAPTER_REGISTRY[name] = cls
        return cls
    return decorator

def create_adapter(model_name: str, config: dict) -> BaseModelAdapter:
    adapter_cls = _ADAPTER_REGISTRY[model_name]
    adapter = adapter_cls(config)
    return adapter
```

### 3. ACT 适配器

```python
@register_adapter("act")
class ACTAdapter(BaseModelAdapter):
    """ACT 模型适配器 — 动作块缓存"""

    def load(self, checkpoint_path: str):
        checkpoint = torch.load(checkpoint_path, map_location=self.device)
        self.model = ACTModel(action_dim=7, state_dim=7, chunk_len=50)
        self.model.load_state_dict(checkpoint["model_state_dict"])
        self.model.to(self.device).eval()
        self._action_buffer = []

    def predict(self, image, instruction, state=None):
        # 每 N 步推理一次，缓存动作块
        if len(self._action_buffer) == 0:
            with torch.no_grad():
                image_t = preprocess(image).to(self.device)
                state_t = torch.tensor(state, dtype=torch.float32).to(self.device)
                chunk = self.model(image_t.unsqueeze(0), state_t.unsqueeze(0))
            self._action_buffer = chunk[0].cpu().numpy().tolist()

        action = np.array(self._action_buffer.pop(0))
        return self._clip_action(action)
```

### 4. OpenVLA 适配器

```python
@register_adapter("openvla")
class OpenVLAAdapter(BaseModelAdapter):
    """OpenVLA 适配器 — 7D 离散 token 解码"""

    def predict(self, image, instruction, state=None):
        # 推理获取 token
        inputs = self.processor(images=image, text=instruction, return_tensors="pt")
        with torch.no_grad():
            outputs = self.model.generate(**inputs)
        # 解码 7D 动作
        bins = decode_tokens(outputs, num_bins=256)
        action = decode_action_7d(bins)
```

### 5. RandomAdapter（测试用）

```python
@register_adapter("random")
class RandomAdapter(BaseModelAdapter):
    """随机动作适配器 — 用于集成测试"""

    def load(self, checkpoint_path: str):
        pass  # 无需加载

    def predict(self, image, instruction, state=None):
        action = np.random.uniform(self.action_low, self.action_high)
        # 倾向于更多张开夹爪（模拟探索）
        action[6] = np.random.choice([0.0, 1.0], p=[0.3, 0.7])
        return action

    def unload(self):
        pass
```

### 6. 适配器管理器

```python
class AdapterManager:
    """管理模型加载/切换/推理"""

    def __init__(self):
        self.current_adapter: BaseModelAdapter | None = None

    def load_model(self, model_name: str, checkpoint_path: str, **config):
        if self.current_adapter:
            self.current_adapter.unload()
        config["checkpoint_path"] = checkpoint_path
        adapter = create_adapter(model_name, config)
        adapter.load(checkpoint_path)
        self.current_adapter = adapter

    def predict(self, image, instruction, state=None) -> np.ndarray:
        """返回 (7,) 动作"""
        if self.current_adapter is None:
            raise RuntimeError("No model loaded")
        return self.current_adapter.predict(image, instruction, state)

    def unload(self):
        if self.current_adapter:
            self.current_adapter.unload()
```

## 验证标准
- [ ] 每种适配器 predict 返回 (7,) shape
- [ ] 返回值在 action_low/action_high 范围内
- [ ] 适配器可加载 checkpoint 并推理
- [ ] RandomAdapter 无需 checkpoint 可直接使用
- [ ] 适配器切换（load→unload→load）正常
- [ ] GPU 显存无泄漏

## 注意事项
- ACT 的动作块缓存策略影响推理平滑性
- OpenVLA 的 token 解码需与 S14 中的编码方式完全对齐
- RandomAdapter 的动作分布应覆盖合理的动作空间
- 图像预处理（归一化、尺寸）各模型可能不同，各自适配

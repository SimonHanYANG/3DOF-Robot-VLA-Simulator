### 2. 数据集读取器

```python
# models/training/utils/dataset.py
class HDF5TrajectoryReader:
    """从 HDF5 文件读取取放轨迹"""

    def __init__(self, h5_path: str):
        self.file = h5py.File(h5_path, "r")
        self.trajectory_ids = [k for k in self.file.keys() if k.startswith("trajectory_")]

    def read(self, trajectory_id: str) -> dict:
        """读取一条轨迹"""
        grp = self.file[trajectory_id]
        return {
            "instruction": grp.attrs["instruction"],
            "object_type": grp.attrs["object_type"],
            "actions": grp["actions"][:],              # (T, 7)
            "ee_poses": grp["ee_poses"][:],            # (T, 7)
            "joint_positions": grp["joint_positions"][:],  # (T, 6)
            "gripper_states": grp["gripper_states"][:],    # (T,)
            "top_images": grp["top_images"][:],        # (T, 224, 224, 3)
            "wrist_images": grp["wrist_images"][:],    # (T, 224, 224, 3)
        }
```

### 3. OpenVLA 格式转换

OpenVLA 使用离散化动作 token。7D 动作的编码策略：
- 6D 连续部分（dx,dy,dz,droll,dpitch,dyaw）：每个维度 256 bins
- 1D 二值部分（gripper）：开/关 2 个 token

```python
def convert_to_openvla(h5_path, output_path, num_bins=256):
    """
    HDF5 → HuggingFace Dataset 格式（OpenVLA 训练用）
    """
    reader = HDF5TrajectoryReader(h5_path)
    samples = []

    for tid in reader.trajectory_ids:
        traj = reader.read(tid)
        for t in range(len(traj["actions"])):
            action = traj["actions"][t]  # (7,)
            # 离散化前 6 维
            action_bins = discretize_6d(action[:6], num_bins)
            # 第 7 维（gripper）二值化
            gripper_bin = 1 if action[6] > 0.5 else 0

            samples.append({
                "image": traj["top_images"][t],
                "instruction": traj["instruction"],
                "action_bins": np.concatenate([action_bins, [gripper_bin]]),
            })

    # 保存为 HuggingFace datasets 格式
    dataset = datasets.Dataset.from_list(samples)
    dataset.save_to_disk(output_path)
```

### 4. ACT 格式转换

ACT 使用 HDF5 格式的动作块：

```python
def convert_to_act(h5_path, output_path, chunk_len=50):
    """
    HDF5 → ACT 格式（chunked actions）
    """
    reader = HDF5TrajectoryReader(h5_path)
    with h5py.File(output_path, "w") as out:
        out.attrs["action_dim"] = 7
        out.attrs["state_dim"] = 7  # ee_pose (7D)
        out.attrs["chunk_len"] = chunk_len

        idx = 0
        for tid in reader.trajectory_ids:
            traj = reader.read(tid)
            T = len(traj["actions"])
            # 滑动窗口提取 chunk
            for t in range(0, T - chunk_len, chunk_len // 2):
                grp = out.create_group(f"sample_{idx:06d}")
                grp.create_dataset("image", data=traj["top_images"][t])
                grp.create_dataset("state", data=traj["ee_poses"][t])  # (7,)
                grp.create_dataset("actions", data=traj["actions"][t:t+chunk_len])  # (chunk, 7)
                idx += 1
```

### 5. π0 格式转换

π0 使用 JSONL 格式：

```python
def convert_to_pi0(h5_path, output_path):
    """
    HDF5 → JSONL 格式（π0 训练用）
    """
    reader = HDF5TrajectoryReader(h5_path)
    with open(output_path, "w") as f:
        for tid in reader.trajectory_ids:
            traj = reader.read(tid)
            # 每条轨迹一个 JSON 对象
            record = {
                "instruction": traj["instruction"],
                "images": {
                    "top": encode_image_list(traj["top_images"]),
                    "wrist": encode_image_list(traj["wrist_images"]),
                },
                "actions": traj["actions"].tolist(),  # [[7], [7], ...]
                "states": traj["joint_positions"].tolist(),  # [[6], [6], ...]
            }
            f.write(json.dumps(record) + "\n")
```

### 6. 转换入口脚本

```python
# scripts/convert_dataset.py
import argparse

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True, help="HDF5 数据集路径")
    parser.add_argument("--format", choices=["openvla", "act", "pi0"], required=True)
    parser.add_argument("--output", required=True, help="输出路径")
    args = parser.parse_args()

    if args.format == "openvla":
        convert_to_openvla(args.input, args.output)
    elif args.format == "act":
        convert_to_act(args.input, args.output)
    elif args.format == "pi0":
        convert_to_pi0(args.input, args.output)
```

## 注意事项
- OpenVLA 离散化的 bin 数量影响动作精度，256 bins 精度约 0.16mm 位置 / 0.78mrad 旋转
- ACT chunk_len 对取放任务建议 30-50 步（取放比推动更结构化）
- π0 JSONL 文件可能很大（图像 base64 编码），考虑使用压缩或外部图像文件
- 转换过程可能耗时数小时，需显示进度条

### 3. 正运动学简化渲染

不使用完整的 DH 变换（太复杂），而是用**简化的几何体**近似：

```tsx
function UR5Robot({ jointAngles, gripperOpen }: Props) {
  const [j1, j2, j3, j4, j5, j6] = jointAngles

  return (
    <group position={[0, 0, 0]}>
      {/* 固定基座 */}
      <Cylinder position={[0, 0, 0.05]} args={[0.06, 0.06, 0.1]} color="#333" />

      {/* J1: shoulder_pan (绕 Z) */}
      <group rotation={[0, 0, j1]}>
        <Cylinder position={[0, 0, 0.1]} args={[0.05, 0.05, 0.05]} color="#444" />

        {/* J2: shoulder_lift (绕 Y) */}
        <group position={[0, 0, 0.138]} rotation={[0, j2, 0]}>
          <Box position={[0, 0, -0.21]} args={[0.06, 0.05, 0.42]} color="#e74c3c" />

          {/* J3: elbow (绕 Y) */}
          <group position={[0, 0, -0.425]} rotation={[0, j3, 0]}>
            <Box position={[0, 0, -0.196]} args={[0.05, 0.04, 0.39]} color="#3498db" />

            {/* J4: wrist_1 (绕 Y) */}
            <group position={[0, 0, -0.392]} rotation={[0, j4, 0]}>
              {/* J5: wrist_2 (绕 Z) */}
              <group rotation={[0, 0, j5]}>
                {/* J6: wrist_3 (绕 X) */}
                <group rotation={[j6, 0, 0]}>
                  {/* 法兰端 */}
                  <Cylinder position={[0, 0, -0.05]} args={[0.03, 0.03, 0.02]} color="#555" />
                  {/* 夹爪 */}
                  <Gripper open={gripperOpen} />
                </group>
              </group>
            </group>
          </group>
        </group>
      </group>
    </group>
  )
}
```

### 4. 物体渲染

```tsx
function Objects({ objPoses }: { objPoses: ObjectPoses }) {
  return (
    <>
      {Object.entries(objPoses).map(([name, pose]) => {
        const position = [pose.pos[0], pose.pos[2], -pose.pos[1]]  // MuJoCo → Three.js
        const quaternion = new THREE.Quaternion(
          pose.quat[1], pose.quat[2], pose.quat[3], pose.quat[0]  // wxyz → xyzw
        )

        if (name.includes("cube")) {
          return <Box key={name} position={position} quaternion={quaternion}
                      args={[0.04, 0.04, 0.04]} color="red" />
        }
        if (name.includes("sphere")) {
          return <Sphere key={name} position={position} quaternion={quaternion}
                         args={[0.025, 16, 16]} color="green" />
        }
        if (name.includes("cylinder")) {
          return <Cylinder key={name} position={position} quaternion={quaternion}
                           args={[0.015, 0.015, 0.06, 16]} color="blue" />
        }
        return null
      })}
    </>
  )
}
```

### 5. 坐标系转换

MuJoCo (Z-up) → Three.js (Y-up)：
```typescript
function mujocoToThree(pos: number[]): [number, number, number] {
  return [pos[0], pos[2], -pos[1]]
}

function mujocoQuatToThree(quat: number[]): THREE.Quaternion {
  // MuJoCo: (w, x, y, z) → Three.js: (x, y, z, w)
  return new THREE.Quaternion(quat[1], quat[2], quat[3], quat[0])
}
```

## 注意事项
- 坐标系转换（MuJoCo Z-up ↔ Three.js Y-up）是常见 bug 源头，需仔细验证
- 四元数顺序 (w,x,y,z) ↔ (x,y,z,w) 不一致会导致物体旋转异常
- 初期使用简化几何体渲染，性能开销低，后续可替换为 STL mesh
- WebSocket 连接失败时显示错误提示而非空白页面

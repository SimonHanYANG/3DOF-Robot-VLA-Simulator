### 2. 物体与目标位置设置

```tsx
function TaskPanel({ onSetObject, onSetTarget }: Props) {
  const [objType, setObjType] = useState("cube")
  const [placingMode, setPlacingMode] = useState<"object" | "target" | null>(null)

  // 点击桌面设置位置
  const handleTableClick = (event) => {
    const point = event.point  // Three.js raycast 坐标
    if (placingMode === "object") {
      onSetObject(objType, [point.x, -point.z, point.y])  // Three.js → MuJoCo
    } else if (placingMode === "target") {
      onSetTarget([point.x, -point.z])
    }
    setPlacingMode(null)
  }

  return (
    <div>
      <select value={objType} onChange={e => setObjType(e.target.value)}>
        <option value="cube">Cube</option>
        <option value="sphere">Sphere</option>
        <option value="cylinder">Cylinder</option>
      </select>
      <button onClick={() => setPlacingMode("object")}>Set Object Position</button>
      <button onClick={() => setPlacingMode("target")}>Set Target Position</button>
    </div>
  )
}
```

### 3. 推理控制

```tsx
function InferenceControls({ ws }: Props) {
  const [modelName, setModelName] = useState("")
  const [speed, setSpeed] = useState(1.0)
  const [running, setRunning] = useState(false)

  return (
    <div>
      <input placeholder="Model checkpoint path" value={modelName} onChange={...} />
      <button onClick={() => ws.send({type: "load_model", model_name: modelName})}>
        Load Model
      </button>
      <button onClick={() => { ws.send({type: "start_inference", speed}); setRunning(true) }}>
        Start Inference
      </button>
      <button onClick={() => { ws.send({type: "stop_inference"}); setRunning(false) }}>
        Stop
      </button>
      <select value={speed} onChange={e => setSpeed(Number(e.target.value))}>
        <option value={0.5}>0.5x</option>
        <option value={1}>1x</option>
        <option value={2}>2x</option>
      </select>
    </div>
  )
}
```

### 4. 观测面板

```tsx
function ObservationPanel({ state }: Props) {
  return (
    <div>
      {/* 相机图像 */}
      <img src={`data:image/png;base64,${state.top_image}`} alt="Top Camera" />
      <img src={`data:image/png;base64,${state.wrist_image}`} alt="Wrist Camera" />

      {/* 末端位姿 */}
      <div>EE: [{state.ee_pose.slice(0,3).map(v => v.toFixed(3)).join(", ")}]</div>
      <div>Gripper: {state.gripper.toFixed(2)}</div>

      {/* 关节角度 */}
      <div>Joints: [{state.joints.map(v => v.toFixed(3)).join(", ")}]</div>

      {/* 距目标距离 */}
      <div>Dist to target: {computeDistance(state).toFixed(3)}m</div>
    </div>
  )
}
```

## 注意事项
- 桌面点击 raycast 需要将 Three.js 坐标转换回 MuJoCo 坐标
- base64 图像每帧更新，浏览器可能闪白，使用 key 或缓存优化
- 模型加载可能耗时数秒，需显示 loading 状态
- 推理运行中禁用物体/目标位置修改（或允许但先停止推理）

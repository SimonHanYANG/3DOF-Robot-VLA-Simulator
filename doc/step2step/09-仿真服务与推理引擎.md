### 1. SimService

```python
class SimService:
    """仿真服务：管理 SimController 生命周期和状态广播"""

    def __init__(self, scene_path, broadcast_fn):
        self.sim = SimController(scene_path)
        self.broadcast = broadcast_fn  # WebSocket 广播函数

    async def set_object(self, msg):
        """设置物体类型和位置"""
        self.sim.reset(msg["object_type"], msg["position"], self.current_target)

    async def set_target(self, msg):
        """设置目标位置"""
        self.current_target = msg["position"]
        self.sim.reset(self.current_obj_type, self.current_obj_pos, msg["position"])

    async def broadcast_loop(self):
        """25Hz 状态广播循环"""
        while self.running:
            state = self.sim.get_state()
            images = self.sim.get_camera_images()
            await self.broadcast({
                "type": "state_update",
                "ee_pose": state["ee_pose"].tolist(),
                "joints": state["joint_positions"].tolist(),
                "gripper": state["gripper_qpos"],
                "obj_poses": serialize_obj_poses(state["obj_poses"]),
                "top_image": encode_base64(images["top"]),
                "wrist_image": encode_base64(images["wrist"]),
            })
            await asyncio.sleep(1.0 / 25)
```

### 2. InferenceEngine

```python
class InferenceEngine:
    """模型推理闭环引擎"""

    def __init__(self, sim_service, adapter_manager):
        self.sim = sim_service.sim
        self.adapter = adapter_manager
        self.running = False

    async def start(self, config):
        """启动推理循环"""
        self.running = True
        speed = config.get("speed", 1.0)
        step_delay = 0.02 / speed  # 50Hz base

        step_count = 0
        while self.running:
            # 1. 获取观测
            obs = self.sim.get_obs()
            image = obs["top_image"]
            state = obs["joint_positions"]

            # 2. 模型推理 → 7D 动作
            action = self.adapter.predict(image, self.instruction, state)

            # 3. 执行动作
            self.sim.step(action)

            # 4. 广播动作信息
            await self.broadcast_action(action, step_count)

            # 5. 检查终止
            done, success, info = self.sim.is_done()
            if done:
                await self.broadcast_done(success, info)
                break

            step_count += 1
            await asyncio.sleep(step_delay)

    async def stop(self):
        self.running = False
```

### 3. ActionSmoother（7D 动作平滑）

```python
class ActionSmoother:
    """对模型输出的 7D 动作进行平滑，避免突变"""

    def __init__(self, alpha=0.3):
        self.alpha = alpha  # EMA 平滑系数
        self.last_action = np.zeros(7)

    def smooth(self, action: np.ndarray) -> np.ndarray:
        """指数移动平均平滑"""
        smoothed = self.alpha * action + (1 - self.alpha) * self.last_action
        self.last_action = smoothed
        return smoothed

    def reset(self):
        self.last_action = np.zeros(7)
```

### 4. 模型适配器管理器接口

推理引擎通过 `AdapterManager` 调用模型：

```python
class AdapterManager:
    def __init__(self):
        self.current_adapter = None

    def load_model(self, model_name, checkpoint_path, **config):
        """加载模型"""
        if self.current_adapter:
            self.current_adapter.unload()
        self.current_adapter = create_adapter(model_name, config)
        self.current_adapter.load(checkpoint_path)

    def predict(self, image, instruction, state=None) -> np.ndarray:
        """返回 7D 动作"""
        if self.current_adapter is None:
            raise RuntimeError("No model loaded")
        return self.current_adapter.predict(image, instruction, state)
```

## 注意事项
- 推理循环在异步线程中运行，避免阻塞 WebSocket 事件循环
- GPU 推理可能阻塞，需使用 `asyncio.to_thread()` 或专用线程池
- 推理过程中用户可能发送 stop_inference，需优雅终止
- 状态广播和推理循环可能同时访问 SimController，需加锁

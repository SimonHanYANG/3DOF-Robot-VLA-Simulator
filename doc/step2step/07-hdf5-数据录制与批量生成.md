### 1. HDF5 数据模式

```
dataset.h5
├── metadata/                      # 全局元数据
│   ├── "total_trajectories"       : int
│   ├── "action_dim"               : 7
│   ├── "object_types"             : ["cube", "sphere", "cylinder"]
│   ├── "image_size"               : [224, 224]
│   └── "joint_count"              : 6
├── trajectory_000000/             # 每条轨迹一个 group
│   ├── "instruction"              : str  ("pick the red cube and place it at (0.20, 0.15)")
│   ├── "object_type"              : str
│   ├── "object_color"             : str
│   ├── "object_init_pose"         : (7,) float32   # pos(3) + quat(4)
│   ├── "target_pose"              : (3,) float32
│   ├── "num_steps"                : int
│   ├── "grasp_success"            : bool
│   ├── "placement_error"          : float32
│   ├── "actions"                  : (T, 7) float32  # 7D 动作序列
│   ├── "ee_poses"                 : (T, 7) float32  # 末端位姿序列
│   ├── "joint_positions"          : (T, 6) float32  # 关节角度序列
│   ├── "gripper_states"           : (T,) float32    # 夹爪状态序列
│   ├── "top_images"               : (T, 224, 224, 3) uint8  # gzip 压缩
│   └── "wrist_images"             : (T, 224, 224, 3) uint8  # gzip 压缩
├── trajectory_000001/
└── ...
```

### 2. 轨迹写入器

```python
class TrajectoryWriter:
    """HDF5 轨迹写入器"""

    def __init__(self, h5_path: str, mode: str = "a"):
        self.file = h5py.File(h5_path, mode)
        self.count = self._get_existing_count()

    def write(self, trajectory: dict) -> str:
        """写入一条轨迹，返回 trajectory_id"""
        group_name = f"trajectory_{self.count:06d}"
        grp = self.file.create_group(group_name)

        # 标量属性
        grp.attrs["instruction"] = trajectory["instruction"]
        grp.attrs["object_type"] = trajectory["object_type"]
        grp.attrs["grasp_success"] = trajectory["grasp_success"]
        grp.attrs["placement_error"] = trajectory["placement_error"]

        # 数组数据
        grp.create_dataset("actions", data=trajectory["actions"])
        grp.create_dataset("ee_poses", data=trajectory["ee_poses"])
        # ... 其他数据集

        # 图像用 gzip 压缩
        grp.create_dataset("top_images", data=trajectory["top_images"],
                          compression="gzip", compression_opts=4)

        self.count += 1
        self.file.flush()
        return group_name

    def close(self):
        # 更新 metadata
        self.file["metadata"].attrs["total_trajectories"] = self.count
        self.file.close()
```

### 3. 批量生成管道

```python
class DataGenerationPipeline:
    """批量取放轨迹数据生成"""

    def __init__(self, scene_path, output_path, total_trajectories=5000):
        self.sim = SimController(scene_path)
        self.grasp_planner = GraspPlanner(self.sim.model, self.sim.data)
        self.trajectory_gen = PickPlaceTrajectoryGenerator(self.sim, self.grasp_planner)
        self.writer = TrajectoryWriter(output_path)

    def run(self):
        while self.writer.count < self.total:
            # 1. 采样任务
            task = self._sample_task()

            # 2. 生成抓取位姿
            grasp = self.grasp_planner.plan_grasp(task.obj_type, task.obj_pos, task.obj_size)
            if grasp is None:
                continue

            # 3. 生成取放轨迹
            actions = self.trajectory_gen.generate(task.obj_type, task.obj_pos, task.target_pos)
            if actions is None:
                continue

            # 4. 重播验证
            result = self._replay_and_record(actions, task)
            if not result["success"]:
                continue

            # 5. 写入 HDF5
            self.writer.write(result["data"])
```

### 4. 任务采样策略

- **物体类型**：均匀分布（cube 33% / sphere 33% / cylinder 33%）
- **物体颜色**：红/绿/蓝/黄/白 随机
- **物体位置**：桌面工作空间内随机采样（避开边缘 5cm）
- **目标位置**：距物体 10-40cm，桌面工作空间内
- **指令生成**：模板 `"pick the {color} {shape} and place it at ({x:.2f}, {y:.2f})"`

### 5. 断点续传与进度报告

- 检查 HDF5 中已有轨迹数量，从断点继续
- 每 100 条打印统计：总数/成功率/平均步数/平均放置误差
- 每 500 条 flush 文件

- Windows 下 HDF5 文件路径使用 pathlib.Path

## 实现细节

### 1. 末端轨迹曲线（渐变色）

```tsx
function EndpointTrajectory({ points }: { points: Vector3[] }) {
  const colors = useMemo(() => {
    return points.map((_, i) => {
      const t = i / Math.max(points.length - 1, 1)
      return new THREE.Color(0.2 + t * 0.3, 0.4 + t * 0.6, 1.0 - t * 0.5)
    })
  }, [points])

  if (points.length < 2) return null
  return <Line points={points} lineWidth={2} vertexColors={colors} />
}
```

- 蓝（早期）→ 青 → 绿（近期）
- 仅保留最近 200 个点
- 在物体上方 0.005m 处绘制，避免 Z-fighting

### 2. 物体运动轨迹（虚线）

```tsx
function ObjectTrajectory({ points }: { points: Vector3[] }) {
  return (
    <Line points={points} color="orange" lineWidth={1.5}
          dashed dashSize={0.02} gapSize={0.01} />
  )
}
```

### 3. 动作方向箭头

```tsx
function ActionArrow({ origin, direction, scale = 50 }: Props) {
  // direction = (dx, dy, dz) from model output
  // scale up 50x for visibility (0.01m → 0.5m in scene)
  const target = origin.clone().add(direction.clone().multiplyScalar(scale))

  return (
    <group>
      <Cylinder args={[0.002, 0.002, direction.length() * scale, 8]} ...>
        <meshStandardMaterial color="limegreen" />
      </Cylinder>
      <Cone args={[0.006, 0.015, 8]} position={target.toArray()}>
        <meshStandardMaterial color="limegreen" />
      </Cone>
    </group>
  )
}
```

```tsx
function ResultOverlay({ result, finalError }: Props) {
  if (!result) return null
  return (
    <div className={`result-overlay ${result}`}>
      {result === "success" ? (

      ) : (
        <>Failed: {failureReason}</>
      )}
    </div>
  )
}
```
